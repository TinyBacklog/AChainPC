# M2 Pipeline — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `tools/acre.py setup/export/build/run/roots` leva da base NSP, do update NSP e
das keys até um `astralchain-static` que corre 120 s em headless `--recomp-strict` sem
fallbacks.

**Architecture:**
- `setup` usa o próprio `astralchain` (só JIT) para duas coisas:
  - instalar o update na NAND de um perfil isolado (`--install`, flag nova);
  - despejar os segmentos efetivos dos NSO (`--dump-recomp-inputs`, que já existe).
  Depois valida os hashes contra `game/version.json`.
- `export` faz a descoberta de roots (portada de `build_game.py` para
  `tools/acre/roots.py`) e chama o gerador C++ `export_module`, que passa a ser um target
  CMake.
- `build` configura um segundo build tree com `ASTRALCHAIN_RECOMP_DIR=work/export`.
- `run --validate` corre o modo headless strict e lê o JSON de `--recomp-stats`.

**Tech Stack:** Python 3.14 só com a stdlib (`unittest`, `json`, `hashlib`, `subprocess`),
C++20 (frontend e gerador), CMake e Ninja.

**Spec:** `docs/superpowers/specs/2026-10-14-astralchainre-design.md` (secções 6 e 9;
M2 na secção 10).

import hashlib, json, tempfile, unittest
from pathlib import Path
from acre import identity

def make_module(root, name, text=b"t", rodata=b"r", data=b"d"):
    d = root / name / "data"; d.mkdir(parents=True)
    for seg, content in (("text", text), ("rodata", rodata), ("data", data)):
        (d / f"{seg}.bin").write_bytes(content)

def h(b): return hashlib.sha256(b).hexdigest()

class IdentityTest(unittest.TestCase):
    def setUp(self):
        self.tmp = Path(tempfile.mkdtemp())
        self.expected = {"modules": {"main": [h(b"t"), h(b"r"), h(b"d")]},
                         "known_other_versions": {h(b"old")[:12]: "1.0.0"}}

    def test_matching_inputs_pass(self):
        make_module(self.tmp, "main")
        identity.verify(self.tmp, self.expected)

    def test_patched_segment_names_module_and_segment(self):
        make_module(self.tmp, "main", text=b"patched")
        with self.assertRaisesRegex(identity.IdentityError, r"main.*text"):
            identity.verify(self.tmp, self.expected)

    def test_known_old_version_says_update_missing(self):
        make_module(self.tmp, "main", text=b"old")
        with self.assertRaisesRegex(identity.IdentityError, r"1\.0\.0.*update"):
            identity.verify(self.tmp, self.expected)

    def test_missing_module(self):
        with self.assertRaisesRegex(identity.IdentityError, r"main.*missing"):
            identity.verify(self.tmp, self.expected)

    def test_shipped_version_file_is_complete(self):
        exp = identity.load_expected(Path(__file__).parents[3] / "game/version.json")
        self.assertEqual(set(exp["modules"]), {"rtld", "main", "sdk"})

from acre import roots

def write(d, text, rodata=b"", data=b""):
    (d / "data").mkdir(parents=True)
    for n, b in (("text", text), ("rodata", rodata), ("data", data)):
        (d / "data" / f"{n}.bin").write_bytes(b)

class RootsTest(unittest.TestCase):
    def test_code_pointer_in_data_becomes_root(self):
        d = Path(tempfile.mkdtemp())
        text = b"\x1f\x20\x03\xd5" * 64            # 64 NOPs
        write(d, text, rodata=struct.pack("<Q", 0x40), data=struct.pack("<Q", 0x2000))
        found, _ = roots.discover(d, {"text_vaddr": 0, "rodata_vaddr": 0x1000, "data_vaddr": 0x2000})

        self.assertNotIn(0x2000, found)           # not inside .text

        write(d, b"\x1f\x20\x03\xd5" * 64, rodata=struct.pack("<Q", 0x42))
        found, _ = roots.discover(d, {"text_vaddr": 0, "rodata_vaddr": 0x1000, "data_vaddr": 0x2000})
        self.assertNotIn(0x42, found)
